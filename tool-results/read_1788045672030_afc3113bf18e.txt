   152→function useCountUp(target: number, duration = 900): number {
   153→  const [value, setValue] = useState(0)
   154→  useEffect(() => {
   155→    if (target <= 0) return
   156→    let raf = 0
   157→    const start = performance.now()
   158→    const tick = (now: number) => {
   159→      const p = Math.min(1, (now - start) / duration)
   160→      const eased = 1 - Math.pow(1 - p, 3)
   161→      setValue(Math.round(target * eased))
   162→      if (p < 1) raf = requestAnimationFrame(tick)
   163→    }
   164→    raf = requestAnimationFrame(tick)
   165→    return () => cancelAnimationFrame(raf)
   166→  }, [target, duration])
   167→  return value
   168→}

   176→const KPI_ACCENTS: Record<string, KpiAccent> = {

   183→function KpiCard({
   184→  icon: Icon,
   185→  label,
   186→  value,
   187→  suffix,
   188→  hint,
   189→  accentKey,
   190→  delay,
   191→}: {
   192→  icon: LucideIcon
   193→  label: string
   194→  value: number
   195→  suffix?: string
   196→  hint?: string
   197→  accentKey: keyof typeof KPI_ACCENTS

   199→}) {
   200→  const count = useCountUp(value)
   201→  const accent = KPI_ACCENTS[accentKey]

   203→    <motion.div
   204→      initial={{ opacity: 0, y: 14 }}
   205→      animate={{ opacity: 1, y: 0 }}
   206→      transition={{ duration: 0.4, delay }}
   207→    >
   208→      <Card className="group relative overflow-hidden transition-colors hover:border-primary/40">
   209→        <div
   210→          className={cn(
   211→            'absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent to-transparent',
   212→            accent.gradient
   213→          )}
   214→          aria-hidden
   215→        />
   216→        <CardContent className="flex items-start justify-between gap-3 p-5">
   217→          <div className="min-w-0">
   218→            <p className="text-xs font-medium tracking-wide text-muted-foreground uppercase">
   219→              {label}
   220→            </p>

   224→            </p>

   226→          </div>
