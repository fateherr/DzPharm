   530→export function normalizeKey(s: string): string {
   531→  return s
   532→    .normalize('NFD')
   533→    .replace(/[\u0300-\u036f]/g, '')
   534→    .toUpperCase()
   535→    .replace(/[^A-Z0-9 ]/g, ' ')
   536→    .replace(/\s+/g, ' ')
   537→    .trim()
   538→}

   540→/**
   541→ * Détecte les règles applicables sur une liste de médicaments.
   542→ * Chaque médicament est identifié par sa marque (`name` ou `input`) et sa DCI
   543→ * (l'une des deux peut manquer) ; la correspondance se fait sur les jetons
   544→ * DCI/marque normalisés.
   545→ */
   546→export function matchInteractions(
   547→  drugs: { name?: string; input?: string; dci?: string | null }[]
   548→): LocalRule[] {
   549→  const keysPerDrug = drugs.map((d) => {
   550→    const label = d.name ?? d.input ?? ''
   551→    const keys = new Set<string>()
   552→    if (d.dci) {
   553→      const k = normalizeKey(d.dci)
   554→      if (k) keys.add(k)
   555→      // les DCI composées (« X + Y ») sont éclatées
   556→      for (const part of k.split(/(?: \+ | ET )/)) {
   557→        const p = part.trim()
   558→        if (p.length >= 3) keys.add(p)
   559→      }
   560→    }
   561→    const n = normalizeKey(label)
   562→    if (n) keys.add(n)
   563→    return Array.from(keys)
   564→  })
   565→
   566→  const allKeys = new Set(keysPerDrug.flat())
   567→
   568→  return LOCAL_RULES.filter((rule) => {
   569→    const leftHit = rule.left.some((t) => tokenMatch(allKeys, t))
   570→    const rightHit = rule.right.some((t) => tokenMatch(allKeys, t))
   571→    // les deux membres doivent être touchés par des médicaments distincts
   572→    if (!leftHit || !rightHit) return false
   573→    for (let i = 0; i < drugs.length; i++) {
   574→      const di = keysPerDrug[i]
   575→      const leftSelf = rule.left.some((t) => tokenMatch(new Set(di), t))
   576→      const rightSelf = rule.right.some((t) => tokenMatch(new Set(di), t))
   577→      if (leftSelf && rightSelf) {
   578→        // un seul produit couvre les deux membres (ex : association fixe) → non pertinent
   579→        continue
   580→      }
   581→      for (let j = i + 1; j < drugs.length; j++) {
   582→        const dj = keysPerDrug[j]
   583→        const pairLeft = rule.left.some((t) => tokenMatch(new Set(di), t))
   584→        const pairRight = rule.right.some((t) => tokenMatch(new Set(dj), t))
   585→        const pairLeftR = rule.right.some((t) => tokenMatch(new Set(di), t))
   586→        const pairRightL = rule.left.some((t) => tokenMatch(new Set(dj), t))
   587→        if ((pairLeft && pairRight) || (pairLeftR && pairRightL)) return true
   588→      }
   589→    }
   590→    return false
   591→  })
   592→}
   593→
   594→function tokenMatch(keys: Set<string>, token: string): boolean {
   595→  if (keys.has(token)) return true
   596→  const t = ` ${token} `
   597→  for (const k of keys) {
   598→    if (k === token) return true
   599→    if ((` ${k} `).includes(t) && token.length >= 4) return true
   600→    if ((` ${token} `).includes(` ${k} `) && k.length >= 4) return true
   601→  }
   602→  return false
   603→}

   605→export const SEVERITY_WEIGHT: Record<LocalSeverity, number> = {
   606→  'CONTRE-INDIQUE': 4,
   607→  MAJEURE: 3,
   608→  MODEREE: 2,
   609→  MINEURE: 1,
   610→}
   611→
   612→export function globalRiskFromPairs(
   613→  pairs: { severity: LocalSeverity }[]
   614→): 'FAIBLE' | 'MODERE' | 'ELEVE' | 'CRITIQUE' {
   615→  if (pairs.length === 0) return 'FAIBLE'
   616→  const max = Math.max(...pairs.map((p) => SEVERITY_WEIGHT[p.severity]))
   617→  if (max >= 4) return 'CRITIQUE'
   618→  if (max === 3) return 'ELEVE'
   619→  if (max === 2) return 'MODERE'
   620→  return 'FAIBLE'
   621→}

   623→/** Associe une règle aux paires de produits concernés (marques lisibles). */
   624→export function pairsFromRule(
   625→  rule: LocalRule,
   626→  drugs: { name?: string; input?: string; dci?: string | null }[]
   627→): [string, string][] {
   628→  const result: [string, string][] = []
   629→  for (let i = 0; i < drugs.length; i++) {
   630→    for (let j = i + 1; j < drugs.length; j++) {
   631→      const ki = new Set(
   632→        [drugs[i].dci, drugs[i].name ?? drugs[i].input]
   633→          .filter(Boolean)
   634→          .map((s) => normalizeKey(s as string))
   635→      )
   636→      const kj = new Set(
   637→        [drugs[j].dci, drugs[j].name ?? drugs[j].input]
   638→          .filter(Boolean)
   639→          .map((s) => normalizeKey(s as string))
   640→      )
   641→      const iL = rule.left.some((t) => tokenMatch(ki, t))
   642→      const iR = rule.right.some((t) => tokenMatch(ki, t))
   643→      const jL = rule.left.some((t) => tokenMatch(kj, t))
   644→      const jR = rule.right.some((t) => tokenMatch(kj, t))
   645→      if ((iL && jR) || (iR && jL)) {
   646→        const label = (d: { name?: string; input?: string; dci?: string | null }) =>
   647→          d.name ?? d.input ?? d.dci ?? '—'
   648→        result.push([label(drugs[i]), label(drugs[j])])
   649→      }
   650→    }
   651→  }
   652→  return result
   653→}
