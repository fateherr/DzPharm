   107→   107→    queryKey: ['drug', sheetDrugId],

   112→   112→  const drug = data?.drug
   113→   113→  const equivalents = data?.equivalents ?? []

   116→   116→  // Comparateur de prix sur les équivalents référencés en officine

   118→   118→  const minEquivalentPrice =
   119→   119→    pricedEquivalents.length > 0
   120→   120→      ? Math.min(...pricedEquivalents.map((e) => e.price as number))
   121→   121→      : null
   122→   122→  const cheapestEquivalent =
   123→   123→    pricedEquivalents.length > 0
   124→   124→      ? pricedEquivalents.reduce((a, b) => ((a.price ?? Infinity) <= (b.price ?? Infinity) ? a : b))
   125→   125→      : null

   130→   130→    if (!drug) return
   131→   131→    pushRecent({ id: drug.id, brand: drug.brand, dci: drug.dci, status: drug.status })

   133→   133→      countedRef.current = drug.id
   134→   134→      postDrugView(drug.id)
   135→   135→    }
   136→   136→  }, [drug, pushRecent])
   137→   137→
   138→   138→  function handleToggleFavorite() {
   139→   139→    if (!drug) return
   140→   140→    const res = toggleFavorite({ id: drug.id, brand: drug.brand, dci: drug.dci })
   141→   141→    if (res === 'added') {
   142→   142→      toast({
   143→   143→        title: 'Ajouté aux favoris',
   144→   144→        description: `${drug.brand} est accessible depuis l'accueil —Mes favoris.`,

   147→   147→      toast({ title: 'Retiré des favoris', description: `${drug.brand} n'est plus dans vos favoris.` })
   148→   148→    } else {
   149→   149→      toast({
   150→   150→        title: 'Favoris complets',
   151→   151→        description: `Limite de ${MAX_FAVORITES} favoris atteinte.`,
   152→   152→        variant: 'destructive',
   153→   153→      })
   154→   154→    }

   160→   160→
   161→   161→  async function handleShare() {
   162→   162→    if (!drug) return
   163→   163→    const pharmacy = drug.pharmacy?.[0]
   164→   164→    const text = [
   165→   165→      `${drug.brand} — ${drug.dci}`,
   166→   166→      `Statut : ${drug.status === 'ACTIF' ? 'Actif' : drug.status === 'RETRIE' ? 'Retiré du marché' : 'Non renouvelé'}`,

   170→   170→      drug.regNumber ? `AMM : ${drug.regNumber}` : '',
   171→   171→      pharmacy?.ppa != null ? `Prix public : ${formatPrice(pharmacy.ppa)}` : '',
   172→   172→      `Voir la fiche complète sur DzPharm.`,
   173→   173→    ]
   174→   174→      .filter(Boolean)
   175→   175→      .join('\n')
   176→   176→
   177→   177→    let shared = false
   178→   178→    try {
   179→   179→      if (navigator.share) {
   180→   180→        await navigator.share({ title: `${drug.brand} — DzPharm`, text })
   181→   181→        shared = true
   182→   182→      }
   183→   183→    } catch {
   184→   184→      /* partage natif annulé — repli presse-papier */
   185→   185→    }
   186→   186→    if (!shared) {
   187→   187→      let copied = false
   188→   188→      try {
   189→   189→        await navigator.clipboard.writeText(text)
   190→   190→        copied = true
   191→   191→      } catch {
   192→   192→        // Repli execCommand (contextes sans permission Clipboard API)
   193→   193→        try {
   194→   194→          const ta = document.createElement('textarea')
   195→   195→          ta.value = text
   196→   196→          ta.style.position = 'fixed'
   197→   197→          ta.style.opacity = '0'
   198→   198→          document.body.appendChild(ta)
   199→   199→          ta.select()
   200→   200→          copied = document.execCommand('copy')
   201→   201→          document.body.removeChild(ta)
   202→   202→        } catch {
   203→   203→          copied = false
   204→   204→        }
   205→   205→      }
   206→   206→      if (copied) {
   207→   207→        toast({
   208→   208→          title: 'Fiche copiée',
   209→   209→          description: 'Le résumé du médicament est dans le presse-papier.',
   210→   210→        })
   211→   211→      } else {
   212→   212→        toast({
   213→   213→          title: 'Copie impossible',
   214→   214→          description: 'Votre navigateur bloque le presse-papier.',
   215→   215→          variant: 'destructive',
   216→   216→        })
   217→   217→      }
   218→   218→    }
   219→   219→  }
