    54→function useCountUp(target: number, duration = 900): number {
    55→  const [value, setValue] = useState(0)
    56→  useEffect(() => {
    57→    if (target <= 0) return
    58→    let raf = 0
    59→    const start = performance.now()
    60→    const tick = (now: number) => {
    61→      const p = Math.min(1, (now - start) / duration)
    62→      const eased = 1 - Math.pow(1 - p, 3)
    63→      setValue(Math.round(target * eased))
    64→      if (p < 1) raf = requestAnimationFrame(tick)
    65→    }
    66→    raf = requestAnimationFrame(tick)
    67→    return () => cancelAnimationFrame(raf)
    68→  }, [target, duration])
    69→  return value
    70→}

    85→function KpiCard({
    86→  icon: Icon,
    87→  label,
    88→  value,
    89→  suffix,
    90→  hint,
    91→  accentKey,
    92→  delay,

   111→        <div
   112→          className={cn(

   116→          aria-hidden
   117→        />
