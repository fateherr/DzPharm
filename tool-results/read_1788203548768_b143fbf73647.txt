    61→interface TipPayloadItem {
    62→  name?: string | number
    63→  value?: string | number
    64→  payload?: { name?: string; count?: number }
    65→}
    66→
    67→function ChartTip({
    68→  active,
    69→  payload,
    70→  label,
    71→}: {
    72→  active?: boolean
    73→  payload?: TipPayloadItem[]
    74→  label?: string | number
    75→}) {
    76→  if (!active || !payload || payload.length === 0) return null
    77→  const item = payload[0]
    78→  const name = item.payload?.name ?? (typeof item.name === 'string' ? item.name : undefined)
    79→  const value = item.value
    80→  return (
    81→    <div className="rounded-lg border border-border bg-popover px-3 py-2 shadow-xl">
    82→      <p className="max-w-64 truncate text-xs font-semibold text-foreground">
    83→        {name ?? label}
    84→      </p>
    85→      <p className="text-sm font-bold text-primary tabular-nums">
    86→        {formatNumber(Number(value))} référence{Number(value) !== 1 ? 's' : ''}
    87→      </p>
    88→    </div>
    89→  )
    90→}
    91→
    92→const axisTick = { fill: 'var(--muted-foreground)', fontSize: 11 }
    93→
    94→function truncate(value: unknown, max = 30): string {
    95→  const s = String(value ?? '')
    96→  return s.length > max ? s.slice(0, max - 1) + '…' : s
    97→}

    99→/** Tooltip dédié aux prix (DA). */
   100→function PriceTip({
   101→  active,
   102→  payload,
   103→}: {
   104→  active?: boolean
   105→  payload?: TipPayloadItem[]
   106→}) {
   107→  if (!active || !payload || payload.length === 0) return null
   108→  const item = payload[0]
   109→  const p = item.payload as { name?: string; count?: number; avg?: number | null } | undefined
   110→  return (
   111→    <div className="rounded-lg border border-border bg-popover px-3 py-2 shadow-xl">
   112→      <p className="max-w-64 truncate text-xs font-semibold text-foreground">
   113→        {p?.name ?? String(item.name ?? '')}
   114→      </p>
   115→      {typeof p?.avg === 'number' && p.avg !== null ? (
   116→        <p className="text-sm font-bold text-chifa tabular-nums">
   117→          PPA moyen : {formatPrice(p.avg)}
   118→          {p.count ? (
   119→            <span className="ml-1 text-xs font-normal text-muted-foreground">
   120→              ({formatNumber(p.count)} produits)
   121→            </span>
   122→          ) : null}
   123→        </p>
   124→      ) : (
   125→        <p className="text-sm font-bold text-primary tabular-nums">
   126→          {formatNumber(Number(item.value))} produit{Number(item.value) !== 1 ? 's' : ''}
   127→        </p>
   128→      )}
   129→    </div>
   130→  )
   131→}

   143→function DonutLegend({ items }: { items: LegendItem[] }) {
   144→  const total = items.reduce((sum, i) => sum + i.value, 0) || 1
   145→  return (
   146→    <ul className="flex flex-wrap items-center justify-center gap-x-6 gap-y-2">
   147→      {items.map((item) => (
   148→        <li key={item.name} className="flex items-center gap-2 text-sm">
   149→          <span
   150→            className={`size-2.5 shrink-0 rounded-full ${item.colorClass}`}
   151→            aria-hidden
   152→          />
   153→          <span className="text-muted-foreground">{item.name}</span>
   154→          <span className="font-semibold text-foreground tabular-nums">
   155→            {formatNumber(item.value)}
   156→          </span>
   157→          <span className="text-xs text-muted-foreground tabular-nums">
   158→            ({Math.round((item.value / total) * 100)}%)
   159→          </span>
   160→        </li>
   161→      ))}
   162→    </ul>
   163→  )
   164→}

   205→interface LabEntry {
   206→  lab: string
   207→  totalProducts: number
   208→  actifs: number
   209→  retraites: number
   210→  nonRenouveles: number
   211→  localShare: number

   215→
   216→interface LabsResponse {
   217→  labs: LabEntry[]
   218→  total: number
   219→  page: number
   220→  totalPages: number
   221→  topLabsQuick: Array<{ lab: string; totalProducts: number }>
   222→  topFiltered: { lab: string; totalProducts: number } | null
   223→}
   224→
   225→async function fetchLabs(
   226→  params: { q?: string; page?: number; pageSize?: number },
   227→  signal?: AbortSignal
   228→): Promise<LabsResponse> {
   229→  const sp = new URLSearchParams()
   230→  if (params.q?.trim()) sp.set('q', params.q.trim())
   231→  if (params.page) sp.set('page', String(params.page))
   232→  if (params.pageSize) sp.set('pageSize', String(params.pageSize))
   233→  const qs = sp.toString()
   234→  const res = await fetch(`/api/labs${qs ? `?${qs}` : ''}`, { signal })
   235→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)
   236→  return (await res.json()) as LabsResponse
   237→}

   797→async function fetchWithdrawals(signal?: AbortSignal): Promise<WithdrawalsResponse> {
   798→  const res = await fetch('/api/withdrawals', { signal })
   799→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)
   800→  return (await res.json()) as WithdrawalsResponse

   803→/** Libellés courts lisibles pour les motifs de retrait (nomenclature brute en majuscules). */
   804→const REASON_LABELS: Record<string, string> = {
   805→  "RETRAIT PAR LE MSPRH POUR INTERDICTION D'IMPORTATION": "Interdiction d'importation (MSPRH)",
   806→  'RETRAIT PAR LE DETENTEUR POUR MOTIF COMMERCIAL': 'Motif commercial (détenteur)',

   812→  'ARRET DE COMMERCIALISATION': 'Arrêt de commercialisation',
   813→  'retrait par le laboratoire pour raisons commerciales': 'Motif commercial (laboratoire)',
   814→  'RETRAIT PAR LE DETENTEUR POUR MOTIF DE PHARMACO-VIGILANCE (NOTE N°110 DU 17/02/2010)':
   815→    'Pharmacovigilance (MSPRH, note 110)',

   818→function reasonLabel(reason: string | null | undefined): string {
   819→  if (!reason) return 'Motif non communiqué'
   820→  const mapped = REASON_LABELS[reason]
   821→  if (mapped) return mapped
   822→  return truncate(reason.charAt(0) + reason.slice(1).toLowerCase(), 42)
   823→}
   824→
   825→function reasonBadgeClass(reason: string | null | undefined): string {
   826→  if (!reason) return 'border-border bg-muted/60 text-muted-foreground'
   827→  const r = reason.toUpperCase()
   828→  if (r.includes('MSPRH') || r.includes('PHARMACO')) {
   829→    return 'border-state-danger/30 bg-state-danger/10 text-state-danger'
   830→  }
   831→  if (r.includes('DETENTEUR') || r.includes('LABORATOIRE') || r.includes('COMMERCIAL')) {
   832→    return 'border-state-warning/30 bg-state-warning/10 text-state-warning'
   833→  }
   834→  return 'border-border bg-secondary text-secondary-foreground'
   835→}

  1612→function labPageList(page: number, totalPages: number): (number | '…')[] {
  1613→  if (totalPages <= 7) return Array.from({ length: totalPages }, (_, i) => i + 1)
  1614→  const items: (number | '…')[] = [1]
  1615→  const start = Math.max(2, page - 1)
  1616→  const end = Math.min(totalPages - 1, page + 1)
  1617→  if (start > 2) items.push('…')
  1618→  for (let i = start; i <= end; i++) items.push(i)
  1619→  if (end < totalPages - 1) items.push('…')
  1620→  items.push(totalPages)
  1621→  return items
  1622→}
