    69→interface DrugShortagesResponse {
    70→  reports: Array<{
    71→    id: number
    72→    brand: string
    73→    wilaya: string | null
    74→    createdAt: string
    75→    status: string
    76→  }>
    77→  stats: { total: number; active: number; resolved: number; last48h: number }
    78→}
    79→
    80→async function fetchDrugShortages(
    81→  drugId: number,
    82→  signal?: AbortSignal
    83→): Promise<DrugShortagesResponse> {
    84→  const res = await fetch(`/api/shortages?drugId=${drugId}`, { signal })
    85→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)
    86→  return (await res.json()) as DrugShortagesResponse
    87→}

    89→async function postShortageReport(payload: {
    90→  drugId: number
    91→  brand: string
    92→  dci?: string | null
    93→  wilaya?: string | null
    94→  note?: string | null
    95→}): Promise<void> {
    96→  const res = await fetch('/api/shortages', {
    97→    method: 'POST',
    98→    headers: { 'Content-Type': 'application/json' },
    99→    body: JSON.stringify(payload),
   100→  })
   101→  if (!res.ok) {
   102→    const data = (await res.json().catch(() => null)) as { error?: string } | null
   103→    throw new Error(data?.error ?? 'Échec du signalement')
   104→  }
   105→}

   248→    }

   251→      try {
   252→        await navigator.clipboard.writeText(text)
   253→        copied = true
   254→      } catch {
   255→        // Repli execCommand (contextes sans permission Clipboard API)
   256→        try {
   257→          const ta = document.createElement('textarea')
   258→          ta.value = text
   259→          ta.style.position = 'fixed'
   260→          ta.style.opacity = '0'
   261→          document.body.appendChild(ta)
   262→          ta.select()
   263→          copied = document.execCommand('copy')
   264→          document.body.removeChild(ta)
   265→        } catch {
   266→          copied = false
   267→        }
   268→      }
   269→      if (copied) {
   270→        toast({
   271→          title: 'Fiche copiée',
   272→          description: 'Le résumé du médicament est dans le presse-papier.',

   276→          title: 'Copie impossible',
