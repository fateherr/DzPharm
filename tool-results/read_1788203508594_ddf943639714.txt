   139→   139→function VerifierPanel() {
   140→   140→  const basket = useDzPharm((s) => s.basket)
   141→   141→  const addToBasket = useDzPharm((s) => s.addToBasket)
   142→   142→  const removeFromBasket = useDzPharm((s) => s.removeFromBasket)
   143→   143→  const clearBasket = useDzPharm((s) => s.clearBasket)
   144→   144→  const openDrug = useDzPharm((s) => s.openDrug)
   145→   145→  const { toast } = useToast()
   146→   146→
   147→   147→  const [patientContext, setPatientContext] = useState('')
   148→   148→  const [result, setResult] = useState<InteractionsResponse | null>(null)
   149→   149→  const [analyzedIds, setAnalyzedIds] = useState<string>('')
   150→   150→  const [pending, setPending] = useState(false)
   151→   151→
   152→   152→  const items = useMemo(
   153→   153→    () => basket.map((b) => ({ name: b.brand, dci: b.dci })),
   154→   154→    [basket]
   155→   155→  )
   156→   156→
   157→   157→  async function analyze() {
   158→   158→    if (items.length < 2) return
   159→   159→    setPending(true)
   160→   160→    try {
   161→   161→      // Vérification instantanée (moteur de règles, déterministe) puis
   162→   162→      // analyse approfondie — le verdict des paires couvertes par la base
   163→   163→      // de référence est identique dans les deux cas (arbitrage serveur).
   164→   164→      const local = await postLocalInteractions(items)
   165→   165→      setResult(local)
   166→   166→      setAnalyzedIds(basket.map((b) => b.id).join(','))
   167→   167→
   168→   168→      const ai = await postInteractions(items, patientContext)
   169→   169→      if (ai) setResult(ai)
   170→   170→    } catch {
   171→   171→      // le résultat local instantané reste affiché s'il est disponible
   172→   172→    } finally {
   173→   173→      setPending(false)
   174→   174→    }
   175→   175→  }
   176→   176→  const stale =
   177→   177→    result !== null && analyzedIds !== basket.map((b) => b.id).join(',')
   178→   178→
   179→   179→  const severityCounts = useMemo(() => {
   180→   180→    const counts = new Map<InteractionSeverity, number>()
   181→   181→    for (const p of result?.pairs ?? []) {
   182→   182→      counts.set(p.severity, (counts.get(p.severity) ?? 0) + 1)
   183→   183→    }
   184→   184→    return counts
   185→   185→  }, [result])
   186→   186→
   187→   187→  function handleSelect(name: Parameters<typeof addToBasket>[0]) {
   188→   188→    const res = addToBasket(name)
   189→   189→    if (res === 'added') {
   190→   190→      toast({
   191→   191→        title: 'Médicament ajouté',
   192→   192→        description: `${name.brand} (${name.dci}) — ${basket.length + 1} produit(s) dans le panier.`,
   193→   193→      })
   194→   194→    } else if (res === 'duplicate') {
   195→   195→      toast({ title: 'Déjà présent', description: `${name.brand} est déjà dans le panier.` })
   196→   196→    } else {
   197→   197→      toast({
   198→   198→        title: 'Panier complet',
   199→   199→        description: `Le contrôle est limité à ${MAX_BASKET} médicaments.`,
   200→   200→        variant: 'destructive',
   201→   201→      })
   202→   202→    }
   203→   203→  }
   204→   204→
   205→   205→  const risk = result ? RISK_META[result.globalRisk] : null
   206→   206→
   207→   207→  return (
   208→   208→    <div className="grid grid-cols-1 gap-6 lg:grid-cols-[380px_1fr] xl:grid-cols-[420px_1fr]">
   209→   209→      {/* ------------------------- Panier ------------------------- */}
   210→   210→      <div className="space-y-4">
   211→   211→        <Card>
   212→   212→          <CardHeader className="pb-3">
   213→   213→            <CardTitle className="flex items-center justify-between text-base">
   214→   214→              <span className="flex items-center gap-2">
   215→   215→                <Stethoscope className="size-4 text-primary" aria-hidden />
   216→   216→                Panier d&apos;analyse
   217→   217→              </span>
   218→   218→              <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary tabular-nums">
   219→   219→                {basket.length}/{MAX_BASKET}
   220→   220→              </span>
   221→   221→            </CardTitle>
   222→   222→          </CardHeader>
   223→   223→          <CardContent className="space-y-4">
   224→   224→            <SearchAutocomplete
   225→   225→              placeholder="Ajouter un médicament…"
   226→   226→              ariaLabel="Ajouter un médicament au contrôle d'interactions"
   227→   227→              onSelect={handleSelect}
   228→   228→              rightHint={
   229→   229→                <Plus className="size-4 shrink-0 text-muted-foreground/50" aria-hidden />
   230→   230→              }
   231→   231→            />
   232→   232→
   233→   233→            {basket.length === 0 ? (
   234→   234→              <p className="rounded-lg border border-dashed border-border p-4 text-center text-sm text-muted-foreground">
   235→   235→                Le panier est vide. Recherchez un médicament pour l&apos;ajouter —
   236→   236→                depuis le répertoire, utilisez le bouton «&nbsp;Ajouter au contrôle
   237→   237→                d&apos;interactions&nbsp;».
   238→   238→              </p>
   239→   239→            ) : (
   240→   240→              <ul className="flex flex-wrap gap-2" aria-label="Médicaments sélectionnés">
   241→   241→                {basket.map((item) => (
   242→   242→                  <li key={item.id}>
   243→   243→                    <span className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-secondary/60 py-1.5 pr-1.5 pl-3 text-sm">
   244→   244→                      <button
   245→   245→                        type="button"
   246→   246→                        onClick={() => openDrug(item.id)}
   247→   247→                        className="min-w-0 text-left focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   248→   248→                        title="Ouvrir la fiche du médicament"
   249→   249→                      >
   250→   250→                        <span className="block truncate font-medium text-foreground">
   251→   251→                          {item.brand}
   252→   252→                        </span>
   253→   253→                        <span className="block truncate text-[11px] text-muted-foreground">
   254→   254→                          {item.dci}
   255→   255→                        </span>
   256→   256→                      </button>
   257→   257→                      <button
   258→   258→                        type="button"
   259→   259→                        onClick={() => removeFromBasket(item.id)}
   260→   260→                        aria-label={`Retirer ${item.brand} du panier`}
   261→   261→                        className="flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-state-danger/10 hover:text-state-danger focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   262→   262→                      >
   263→   263→                        <X className="size-3.5" />
   264→   264→                      </button>
   265→   265→                    </span>
   266→   266→                  </li>
   267→   267→                ))}
   268→   268→              </ul>
   269→   269→            )}
   270→   270→
   271→   271→            <div>
   272→   272→              <label
   273→   273→                htmlFor="patient-context"
   274→   274→                className="mb-1.5 block text-xs font-medium text-muted-foreground"
   275→   275→              >
   276→   276→                Contexte patient — âge, pathologies, grossesse…
   277→   277→              </label>
   278→   278→              <Textarea
   279→   279→                id="patient-context"
   280→   280→                value={patientContext}
   281→   281→                onChange={(e) => setPatientContext(e.target.value)}
   282→   282→                placeholder="Ex : femme enceinte 2e trimestre, insuffisance rénale chronique, 72 ans…"
   283→   283→                rows={3}
   284→   284→                className="resize-none text-sm"
   285→   285→              />
   286→   286→            </div>
   287→   287→
   288→   288→            <div className="flex gap-2">
   289→   289→              <Button
   290→   290→                onClick={analyze}
   291→   291→                disabled={basket.length < 2 || pending}
   292→   292→                className="h-11 flex-1 text-sm font-semibold"
   293→   293→              >
   294→   294→                {pending ? (
   295→   295→                  <>
   296→   296→                    <Loader2 className="size-4 animate-spin" aria-hidden />
   297→   297→                    Analyse…
   298→   298→                  </>
   299→   299→                ) : (
   300→   300→                  <>
   301→   301→                    <ShieldAlert className="size-4" aria-hidden />
   302→   302→                    Analyser les interactions
   303→   303→                  </>
   304→   304→                )}
   305→   305→              </Button>
   306→   306→              {basket.length > 0 ? (
   307→   307→                <Button
   308→   308→                  variant="ghost"
   309→   309→                  size="icon"
   310→   310→                  className="size-11 text-muted-foreground hover:text-state-danger"
   311→   311→                  onClick={() => {
   312→   312→                    clearBasket()
   313→   313→                    setResult(null)
   314→   314→                  }}
   315→   315→                  aria-label="Vider le panier"
   316→   316→                >
   317→   317→                  <Trash2 className="size-4" />
   318→   318→                </Button>
   319→   319→              ) : null}
   320→   320→            </div>
   321→   321→            {pending ? (
   322→   322→              <p className="flex items-center justify-center gap-2 text-center text-xs text-muted-foreground">
   323→   323→                <Loader2 className="size-3 animate-spin" aria-hidden />
   324→   324→                Vérification des {basket.length} médicaments en cours…
   325→   325→              </p>
   326→   326→            ) : null}
   327→   327→          </CardContent>
   328→   328→        </Card>
   329→   329→      </div>
   330→   330→
   331→   331→      {/* ------------------------ Résultats ------------------------ */}
   332→   332→      <div className="min-w-0">
   333→   333→        {!result && !pending ? (
   334→   334→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-dashed border-border p-10 text-center">
   335→   335→            <ShieldCheck className="size-12 text-muted-foreground/40" aria-hidden />
   336→   336→            <p className="mt-4 text-base font-semibold text-foreground">
   337→   337→              Aucune analyse pour le moment
   338→   338→            </p>
   339→   339→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   340→   340→              Ajoutez au moins 2 médicaments au panier puis lancez l&apos;analyse. Le
   341→   341→              rapport présente le niveau de risque global, les associations à
   342→   342→              surveiller et les conseils de dispensation.
   343→   343→            </p>
   344→   344→          </div>
   345→   345→        ) : pending && !result ? (
   346→   346→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-border bg-card p-10 text-center">
   347→   347→            <Loader2 className="size-10 animate-spin text-primary" aria-hidden />
   348→   348→            <p className="mt-4 text-base font-semibold text-foreground">
   349→   349→              Analyse en cours…
   350→   350→            </p>
   351→   351→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   352→   352→              Vérification des {basket.length} médicaments du panier.
   353→   353→            </p>
   354→   354→          </div>
   355→   355→        ) : result ? (
   356→   356→          <div className="space-y-4">
   357→   357→            {stale ? (
   358→   358→              <div
   359→   359→                className="flex items-center gap-2 rounded-lg border border-state-warning/40 bg-state-warning/10 px-4 py-2.5 text-sm text-state-warning"
   360→   360→                role="status"
   361→   361→              >
   362→   362→                <AlertTriangle className="size-4 shrink-0" aria-hidden />
   363→   363→                Le panier a été modifié depuis cette analyse — relancez l&apos;analyse
   364→   364→                pour actualiser les résultats.
   365→   365→              </div>
   366→   366→            ) : null}
   367→   367→
   368→   368→            {/* Bandeau risque global */}
   369→   369→            <div
   370→   370→              className={cn(
   371→   371→                'flex items-start gap-4 rounded-xl border p-5',
   372→   372→                risk?.border,
   373→   373→                risk?.className
   374→   374→              )}
   375→   375→              role="status"
   376→   376→              aria-live="polite"
   377→   377→            >
   378→   378→              {result.globalRisk === 'FAIBLE' ? (
   379→   379→                <ShieldCheck className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   380→   380→              ) : (
   381→   381→                <ShieldAlert className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   382→   382→              )}
   383→   383→              <div className="min-w-0">
   384→   384→                <p className="text-lg font-bold tracking-tight">
   385→   385→                  {risk?.label}
   386→   386→                  {result.pairs.length > 0 ? (
   387→   387→                    <span className="ml-2 align-middle text-sm font-medium opacity-80">
   388→   388→                      · {result.pairs.length} association{result.pairs.length > 1 ? 's' : ''} analysée
   389→   389→                      {result.pairs.length > 1 ? 's' : ''}
   390→   390→                    </span>
   391→   391→                  ) : null}
   392→   392→                </p>
   393→   393→                <p className="mt-1 text-sm leading-relaxed text-foreground/90">{result.summary}</p>
   394→   394→              </div>
   395→   395→            </div>
   396→   396→
   397→   397→            {/* Répartition des gravités */}
   398→   398→            {result.pairs.length > 0 ? (
   399→   399→              <div className="flex flex-wrap items-center gap-x-5 gap-y-2 rounded-lg border border-border bg-card px-4 py-2.5">
   400→   400→                {SEVERITY_ORDER.filter((s) => severityCounts.has(s)).map((s) => (
   401→   401→                  <span
   402→   402→                    key={s}
   403→   403→                    className="flex items-center gap-1.5 text-xs font-medium text-muted-foreground"
   404→   404→                  >
   405→   405→                    <span className={cn('size-2 rounded-full', SEVERITY_LEGEND[s])} aria-hidden />
   406→   406→                    {SEVERITY_LABELS[s]}
   407→   407→                    <span className="font-semibold text-foreground tabular-nums">
   408→   408→                      {severityCounts.get(s)}
   409→   409→                    </span>
   410→   410→                  </span>
   411→   411→                ))}
   412→   412→              </div>
   413→   413→            ) : null}
   414→   414→
   415→   415→            {/* Médicaments enrichis */}
   416→   416→            {result.enriched.length > 0 ? (
   417→   417→              <Card>
   418→   418→                <CardHeader className="pb-2">
   419→   419→                  <CardTitle className="text-sm">Produits analysés</CardTitle>
   420→   420→                </CardHeader>
   421→   421→                <CardContent className="flex flex-wrap gap-2">
   422→   422→                  {result.enriched.map((e) => (
   423→   423→                    <span
   424→   424→                      key={e.input}
   425→   425→                      className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-muted/40 px-3 py-1.5 text-xs"
   426→   426→                    >
   427→   427→                      <span className="min-w-0">
   428→   428→                        <span className="block truncate font-semibold text-foreground">
   429→   429→                          {e.input}
   430→   430→                        </span>
   431→   431→                        {e.dci ? (
   432→   432→                          <span className="block truncate text-muted-foreground">{e.dci}</span>
   433→   433→                        ) : null}
   434→   434→                      </span>
   435→   435→                      {e.status ? (
