   128→function useCountUp(target: number, duration = 900): number {
   129→  const [value, setValue] = useState(0)
   130→  useEffect(() => {
   131→    if (target <= 0) return
   132→    let raf = 0
   133→    const start = performance.now()
   134→    const tick = (now: number) => {
   135→      const p = Math.min(1, (now - start) / duration)
   136→      const eased = 1 - Math.pow(1 - p, 3)
   137→      setValue(Math.round(target * eased))
   138→      if (p < 1) raf = requestAnimationFrame(tick)
   139→    }
   140→    raf = requestAnimationFrame(tick)
   141→    return () => cancelAnimationFrame(raf)
   142→  }, [target, duration])
   143→  return value
   144→}

   152→const KPI_ACCENTS: Record<string, KpiAccent> = {

   154→  primary: { text: 'text-primary', bg: 'bg-primary/10', gradient: 'via-primary/60' },

   159→function KpiCard({
   160→  icon: Icon,
   161→  label,
   162→  value,
   163→  suffix,
   164→  hint,
   165→  accentKey,
   166→  delay,
   167→}: {
   168→  icon: LucideIcon
   169→  label: string
   170→  value: number
   171→  suffix?: string
   172→  hint?: string
   173→  accentKey: keyof typeof KPI_ACCENTS

   175→}) {
   176→  const count = useCountUp(value)
   177→  const accent = KPI_ACCENTS[accentKey]

   179→    <motion.div
   180→      initial={{ opacity: 0, y: 14 }}
   181→      animate={{ opacity: 1, y: 0 }}
   182→      transition={{ duration: 0.4, delay }}

   185→        <div
   186→          className={cn(
   187→            'absolute inset-x-0 top-0 h-px bg-gradient-to-r from-transparent to-transparent',
   188→            accent.gradient
   189→          )}
   190→          aria-hidden
   191→        />
   192→        <CardContent className="flex items-start justify-between gap-3 p-5">
   193→          <div className="min-w-0">
