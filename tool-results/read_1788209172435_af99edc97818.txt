   142→function VerifierPanel() {
   143→  const basket = useDzPharm((s) => s.basket)
   144→  const addToBasket = useDzPharm((s) => s.addToBasket)
   145→  const removeFromBasket = useDzPharm((s) => s.removeFromBasket)
   146→  const clearBasket = useDzPharm((s) => s.clearBasket)
   147→  const openDrug = useDzPharm((s) => s.openDrug)
   148→  const { toast } = useToast()
   149→
   150→  const [patientContext, setPatientContext] = useState('')
   151→  const [result, setResult] = useState<InteractionsResponse | null>(null)
   152→  const [analyzedIds, setAnalyzedIds] = useState<string>('')
   153→  const [pending, setPending] = useState(false)
   154→
   155→  const items = useMemo(
   156→    () => basket.map((b) => ({ name: b.brand, dci: b.dci })),
   157→    [basket]
   158→  )
   159→
   160→  async function analyze() {
   161→    if (items.length < 2) return
   162→    setPending(true)
   163→    try {
   164→      // Vérification instantanée (moteur de règles, déterministe) puis
   165→      // analyse approfondie — le verdict des paires couvertes par la base
   166→      // de référence est identique dans les deux cas (arbitrage serveur).
   167→      const local = await postLocalInteractions(items)
   168→      setResult(local)
   169→      setAnalyzedIds(basket.map((b) => b.id).join(','))
   170→
   171→      const ai = await postInteractions(items, patientContext)
   172→      if (ai) setResult(ai)
   173→    } catch {
   174→      // le résultat local instantané reste affiché s'il est disponible
   175→    } finally {
   176→      setPending(false)
   177→    }
   178→  }
   179→  const stale =
   180→    result !== null && analyzedIds !== basket.map((b) => b.id).join(',')
   181→
   182→  const severityCounts = useMemo(() => {
   183→    const counts = new Map<InteractionSeverity, number>()
   184→    for (const p of result?.pairs ?? []) {
   185→      counts.set(p.severity, (counts.get(p.severity) ?? 0) + 1)
   186→    }
   187→    return counts
   188→  }, [result])
   189→
   190→  function handleSelect(name: Parameters<typeof addToBasket>[0]) {
   191→    const res = addToBasket(name)
   192→    if (res === 'added') {
   193→      toast({
   194→        title: 'Médicament ajouté',
   195→        description: `${name.brand} (${name.dci}) — ${basket.length + 1} produit(s) dans le panier.`,
   196→      })
   197→    } else if (res === 'duplicate') {
   198→      toast({ title: 'Déjà présent', description: `${name.brand} est déjà dans le panier.` })
   199→    } else {
   200→      toast({
   201→        title: 'Panier complet',
   202→        description: `Le contrôle est limité à ${MAX_BASKET} médicaments.`,
   203→        variant: 'destructive',
   204→      })
   205→    }
   206→  }
   207→
   208→  const risk = result ? RISK_META[result.globalRisk] : null
   209→
   210→  return (
   211→    <div className="grid grid-cols-1 gap-6 lg:grid-cols-[380px_1fr] xl:grid-cols-[420px_1fr]">
   212→      {/* ------------------------- Panier ------------------------- */}
   213→      <div className="space-y-4">
   214→        <Card>
   215→          <CardHeader className="pb-3">
   216→            <CardTitle className="flex items-center justify-between text-base">
   217→              <span className="flex items-center gap-2">
   218→                <Stethoscope className="size-4 text-primary" aria-hidden />
   219→                Panier d&apos;analyse
   220→              </span>
   221→              <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary tabular-nums">
   222→                {basket.length}/{MAX_BASKET}
   223→              </span>
   224→            </CardTitle>
   225→          </CardHeader>
   226→          <CardContent className="space-y-4">
   227→            <SearchAutocomplete
   228→              placeholder="Ajouter un médicament…"
   229→              ariaLabel="Ajouter un médicament au contrôle d'interactions"
   230→              onSelect={handleSelect}
   231→              rightHint={
   232→                <Plus className="size-4 shrink-0 text-muted-foreground/50" aria-hidden />
   233→              }
   234→            />
   235→
   236→            {basket.length === 0 ? (
   237→              <p className="rounded-lg border border-dashed border-border p-4 text-center text-sm text-muted-foreground">
   238→                Le panier est vide. Recherchez un médicament pour l&apos;ajouter —
   239→                depuis le répertoire, utilisez le bouton «&nbsp;Ajouter au contrôle
   240→                d&apos;interactions&nbsp;».
   241→              </p>
   242→            ) : (
   243→              <ul className="flex flex-wrap gap-2" aria-label="Médicaments sélectionnés">
   244→                {basket.map((item) => (
   245→                  <li key={item.id}>
   246→                    <span className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-secondary/60 py-1.5 pr-1.5 pl-3 text-sm">
   247→                      <button
   248→                        type="button"
   249→                        onClick={() => openDrug(item.id)}
   250→                        className="min-w-0 text-left focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   251→                        title="Ouvrir la fiche du médicament"
   252→                      >
   253→                        <span className="block truncate font-medium text-foreground">
   254→                          {item.brand}
   255→                        </span>
   256→                        <span className="block truncate text-[11px] text-muted-foreground">
   257→                          {item.dci}
   258→                        </span>
   259→                      </button>
   260→                      <button
   261→                        type="button"
   262→                        onClick={() => removeFromBasket(item.id)}
   263→                        aria-label={`Retirer ${item.brand} du panier`}
   264→                        className="flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-state-danger/10 hover:text-state-danger focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   265→                      >
   266→                        <X className="size-3.5" />
   267→                      </button>
   268→                    </span>
   269→                  </li>
   270→                ))}
   271→              </ul>
   272→            )}
   273→
   274→            <div>
   275→              <label
   276→                htmlFor="patient-context"
   277→                className="mb-1.5 block text-xs font-medium text-muted-foreground"
   278→              >
   279→                Contexte patient — âge, pathologies, grossesse…
   280→              </label>
   281→              <Textarea
   282→                id="patient-context"
   283→                value={patientContext}
   284→                onChange={(e) => setPatientContext(e.target.value)}
   285→                placeholder="Ex : femme enceinte 2e trimestre, insuffisance rénale chronique, 72 ans…"
   286→                rows={3}
   287→                className="resize-none text-sm"
   288→              />
   289→            </div>
   290→
   291→            <div className="flex gap-2">
   292→              <Button
   293→                onClick={analyze}
   294→                disabled={basket.length < 2 || pending}
   295→                className="h-11 flex-1 text-sm font-semibold"
   296→              >
   297→                {pending ? (
   298→                  <>
   299→                    <Loader2 className="size-4 animate-spin" aria-hidden />
   300→                    Analyse…
   301→                  </>
   302→                ) : (
   303→                  <>
   304→                    <ShieldAlert className="size-4" aria-hidden />
   305→                    Analyser les interactions
   306→                  </>
   307→                )}
   308→              </Button>
   309→              {basket.length > 0 ? (
   310→                <Button
   311→                  variant="ghost"
   312→                  size="icon"
   313→                  className="size-11 text-muted-foreground hover:text-state-danger"
   314→                  onClick={() => {
   315→                    clearBasket()
   316→                    setResult(null)
   317→                  }}
   318→                  aria-label="Vider le panier"
   319→                >
   320→                  <Trash2 className="size-4" />
   321→                </Button>
   322→              ) : null}
   323→            </div>
   324→            {pending ? (
   325→              <p className="flex items-center justify-center gap-2 text-center text-xs text-muted-foreground">
   326→                <Loader2 className="size-3 animate-spin" aria-hidden />
   327→                Vérification des {basket.length} médicaments en cours…
   328→              </p>
   329→            ) : null}
   330→          </CardContent>
   331→        </Card>
   332→      </div>
   333→
   334→      {/* ------------------------ Résultats ------------------------ */}
   335→      <div className="min-w-0">
   336→        {!result && !pending ? (
   337→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-dashed border-border p-10 text-center">
   338→            <ShieldCheck className="size-12 text-muted-foreground/40" aria-hidden />
   339→            <p className="mt-4 text-base font-semibold text-foreground">
   340→              Aucune analyse pour le moment
   341→            </p>
   342→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   343→              Ajoutez au moins 2 médicaments au panier puis lancez l&apos;analyse. Le
   344→              rapport présente le niveau de risque global, les associations à
   345→              surveiller et les conseils de dispensation.
   346→            </p>
   347→          </div>
   348→        ) : pending && !result ? (
   349→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-border bg-card p-10 text-center">
   350→            <Loader2 className="size-10 animate-spin text-primary" aria-hidden />
   351→            <p className="mt-4 text-base font-semibold text-foreground">
   352→              Analyse en cours…
   353→            </p>
   354→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   355→              Vérification des {basket.length} médicaments du panier.
   356→            </p>
   357→          </div>
   358→        ) : result ? (
   359→          <div className="space-y-4">
   360→            {stale ? (
   361→              <div
   362→                className="flex items-center gap-2 rounded-lg border border-state-warning/40 bg-state-warning/10 px-4 py-2.5 text-sm text-state-warning"
   363→                role="status"
   364→              >
   365→                <AlertTriangle className="size-4 shrink-0" aria-hidden />
   366→                Le panier a été modifié depuis cette analyse — relancez l&apos;analyse
   367→                pour actualiser les résultats.
   368→              </div>
   369→            ) : null}
   370→
   371→            {/* Bandeau risque global */}
   372→            <div
   373→              className={cn(
   374→                'flex items-start gap-4 rounded-xl border p-5',
   375→                risk?.border,
   376→                risk?.className
   377→              )}
   378→              role="status"
   379→              aria-live="polite"
   380→            >
   381→              {result.globalRisk === 'FAIBLE' ? (
   382→                <ShieldCheck className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   383→              ) : (
   384→                <ShieldAlert className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   385→              )}
   386→              <div className="min-w-0">
   387→                <p className="text-lg font-bold tracking-tight">
   388→                  {risk?.label}
   389→                  {result.pairs.length > 0 ? (
   390→                    <span className="ml-2 align-middle text-sm font-medium opacity-80">
   391→                      · {result.pairs.length} association{result.pairs.length > 1 ? 's' : ''} analysée
   392→                      {result.pairs.length > 1 ? 's' : ''}
   393→                    </span>
   394→                  ) : null}
   395→                </p>
   396→                <p className="mt-1 text-sm leading-relaxed text-foreground/90">{result.summary}</p>
   397→              </div>
   398→            </div>
   399→
   400→            {/* Répartition des gravités */}
   401→            {result.pairs.length > 0 ? (
   402→              <div className="flex flex-wrap items-center gap-x-5 gap-y-2 rounded-lg border border-border bg-card px-4 py-2.5">
   403→                {SEVERITY_ORDER.filter((s) => severityCounts.has(s)).map((s) => (
   404→                  <span
   405→                    key={s}
   406→                    className="flex items-center gap-1.5 text-xs font-medium text-muted-foreground"
   407→                  >
   408→                    <span className={cn('size-2 rounded-full', SEVERITY_LEGEND[s])} aria-hidden />
   409→                    {SEVERITY_LABELS[s]}
   410→                    <span className="font-semibold text-foreground tabular-nums">
   411→                      {severityCounts.get(s)}
   412→                    </span>
   413→                  </span>
   414→                ))}
   415→              </div>
   416→            ) : null}
   417→
   418→            {/* Médicaments enrichis */}
   419→            {result.enriched.length > 0 ? (
   420→              <Card>
   421→                <CardHeader className="pb-2">
   422→                  <CardTitle className="text-sm">Produits analysés</CardTitle>
   423→                </CardHeader>
   424→                <CardContent className="flex flex-wrap gap-2">
   425→                  {result.enriched.map((e) => (
   426→                    <span
   427→                      key={e.input}
   428→                      className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-muted/40 px-3 py-1.5 text-xs"
   429→                    >
   430→                      <span className="min-w-0">
   431→                        <span className="block truncate font-semibold text-foreground">
   432→                          {e.input}
   433→                        </span>
   434→                        {e.dci ? (
   435→                          <span className="block truncate text-muted-foreground">{e.dci}</span>
   436→                        ) : null}
   437→                      </span>
   438→                      {e.status ? (
   439→                        e.status === 'RETRIE' ? (
   440→                          <span className="shrink-0 rounded bg-state-danger/15 px-1.5 py-0.5 text-[10px] font-semibold text-state-danger">
   441→                            Retiré
   442→                          </span>
   443→                        ) : (
   444→                          <StatusBadge status={e.status} className="shrink-0" />
   445→                        )
   446→                      ) : null}
   447→                    </span>
   448→                  ))}
   449→                </CardContent>
   450→              </Card>
   451→            ) : null}
   452→
   453→            {/* Paires */}
   454→            {result.pairs.length > 0 ? (
   455→              <section aria-label="Interactions détectées" className="space-y-3">
   456→                <h2 className="text-sm font-semibold tracking-wider text-muted-foreground uppercase">
   457→                  Interactions détectées
   458→                </h2>
   459→                {result.pairs.map((pair, i) => (
   460→                  <PairCard key={`${pair.drugs.join('-')}-${i}`} pair={pair} />
   461→                ))}
   462→              </section>
   463→            ) : (
   464→              <Card className="border-state-safe/40 bg-state-safe/5">
   465→                <CardContent className="flex items-center gap-3 p-4 text-sm text-foreground/90">
   466→                  <ShieldCheck className="size-5 shrink-0 text-state-safe" aria-hidden />
   467→                  Aucune interaction significative identifiée entre les produits analysés.
   468→                </CardContent>
   469→              </Card>
   470→            )}
   471→
   472→            {/* Conseils */}
   473→            {result.advice.length > 0 ? (
   474→              <Card>
   475→                <CardHeader className="pb-2">
   476→                  <CardTitle className="flex items-center gap-2 text-sm">
   477→                    <Lightbulb className="size-4 text-chifa" aria-hidden />
   478→                    Conseils
   479→                  </CardTitle>
   480→                </CardHeader>
   481→                <CardContent>
   482→                  <ul className="list-disc space-y-1.5 pl-5 text-sm text-foreground/90">
   483→                    {result.advice.map((a, i) => (
   484→                      <li key={i}>{a}</li>
   485→                    ))}
   486→                  </ul>
   487→                </CardContent>
   488→              </Card>
   489→            ) : null}
   490→
   491→            {/* Surveillance */}
   492→            {result.monitoring.length > 0 ? (
   493→              <Card>
   494→                <CardHeader className="pb-2">
   495→                  <CardTitle className="flex items-center gap-2 text-sm">
   496→                    <Activity className="size-4 text-state-warning" aria-hidden />
   497→                    Surveillance
   498→                  </CardTitle>
   499→                </CardHeader>
   500→                <CardContent>
   501→                  <ul className="list-disc space-y-1.5 pl-5 text-sm text-foreground/90">
   502→                    {result.monitoring.map((m, i) => (
   503→                      <li key={i}>{m}</li>
   504→                    ))}
   505→                  </ul>
   506→                </CardContent>
   507→              </Card>
   508→            ) : null}
   509→
   510→            {/* Urgences + rappel professionnel au point d'usage (audit 1.9) */}
   511→            <Alert
   512→              variant="default"
   513→              className="border-state-danger/40 bg-state-danger/5 text-state-danger"
   514→              role="note"
   515→              aria-label="En cas d'urgence"
   516→            >
   517→              <Siren className="size-4 shrink-0" aria-hidden />
   518→              <AlertTitle className="text-sm font-semibold">En cas d&apos;urgence</AlertTitle>
   519→              <AlertDescription className="text-xs leading-relaxed text-foreground/90">
   520→                <p>
   521→                  Réaction grave (allergie, surdosage, malaise)&nbsp;: appelez immédiatement le{' '}
   522→                  <a
   523→                    href="tel:14"

   608→function MatrixPanel({ onGoToVerifier }: { onGoToVerifier: () => void }) {

   611→
   612→  const truncated = basket.length > MAX_MATRIX
   613→  const drugs = useMemo(() => basket.slice(0, MAX_MATRIX), [basket])
   614→  const items = useMemo(() => drugs.map((d) => ({ name: d.brand, dci: d.dci })), [drugs])
   615→  const idsKey = drugs.map((d) => d.id).join('|')
   616→
   617→  const { data, isFetching } = useQuery({
   618→    queryKey: ['interactions-matrix', idsKey],
   619→    queryFn: () => postLocalInteractions(items),
   620→    enabled: drugs.length >= 2,
   621→    staleTime: 30 * 1000,
   622→  })
   623→
   624→  /** Paire d'interaction entre deux marques (null si aucune connue). */

   626→    (data?.pairs ?? []).find(
   627→      (p) =>

   629→    )

   632→    return (
   633→      <div className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border p-10 text-center">
