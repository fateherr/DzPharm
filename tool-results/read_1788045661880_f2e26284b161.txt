    41→/* Tooltip personnalisé                                                */
    42→/* ------------------------------------------------------------------ */
    43→
    44→interface TipPayloadItem {
    45→  name?: string | number
    46→  value?: string | number
    47→  payload?: { name?: string; count?: number }
    48→}
    49→
    50→function ChartTip({
    51→  active,
    52→  payload,
    53→  label,
    54→}: {
    55→  active?: boolean
    56→  payload?: TipPayloadItem[]
    57→  label?: string | number
    58→}) {
    59→  if (!active || !payload || payload.length === 0) return null
    60→  const item = payload[0]
    61→  const name = item.payload?.name ?? (typeof item.name === 'string' ? item.name : undefined)
    62→  const value = item.value
    63→  return (
    64→    <div className="rounded-lg border border-border bg-popover px-3 py-2 shadow-xl">
    65→      <p className="max-w-64 truncate text-xs font-semibold text-foreground">
    66→        {name ?? label}
    67→      </p>
    68→      <p className="text-sm font-bold text-primary tabular-nums">
    69→        {formatNumber(Number(value))} référence{Number(value) !== 1 ? 's' : ''}
    70→      </p>
    71→    </div>
    72→  )
    73→}
    74→
    75→const axisTick = { fill: 'var(--muted-foreground)', fontSize: 11 }
    76→
    77→function truncate(value: unknown, max = 30): string {
    78→  const s = String(value ?? '')
    79→  return s.length > max ? s.slice(0, max - 1) + '…' : s
    80→}
    81→
    82→/** Tooltip dédié aux prix (DA). */
    83→function PriceTip({
    84→  active,
    85→  payload,
    86→}: {
    87→  active?: boolean
    88→  payload?: TipPayloadItem[]
    89→}) {
    90→  if (!active || !payload || payload.length === 0) return null
    91→  const item = payload[0]
    92→  const p = item.payload as { name?: string; count?: number; avg?: number | null } | undefined
    93→  return (
    94→    <div className="rounded-lg border border-border bg-popover px-3 py-2 shadow-xl">
    95→      <p className="max-w-64 truncate text-xs font-semibold text-foreground">
    96→        {p?.name ?? String(item.name ?? '')}
    97→      </p>
    98→      {typeof p?.avg === 'number' && p.avg !== null ? (
    99→        <p className="text-sm font-bold text-chifa tabular-nums">
   100→          PPA moyen : {formatPrice(p.avg)}
   101→          {p.count ? (
   102→            <span className="ml-1 text-xs font-normal text-muted-foreground">
   103→              ({formatNumber(p.count)} produits)
   104→            </span>
   105→          ) : null}
   106→        </p>
   107→      ) : (
   108→        <p className="text-sm font-bold text-primary tabular-nums">
   109→          {formatNumber(Number(item.value))} produit{Number(item.value) !== 1 ? 's' : ''}
   110→        </p>
   111→      )}
   112→    </div>
   113→  )
   114→}

   117→/* Légende personnalisée pour les donuts                               */
   118→/* ------------------------------------------------------------------ */
   119→
   120→interface LegendItem {
   121→  name: string
   122→  value: number
   123→  colorClass: string
   124→}
   125→
   126→function DonutLegend({ items }: { items: LegendItem[] }) {
   127→  const total = items.reduce((sum, i) => sum + i.value, 0) || 1
   128→  return (
   129→    <ul className="flex flex-wrap items-center justify-center gap-x-6 gap-y-2">
   130→      {items.map((item) => (
   131→        <li key={item.name} className="flex items-center gap-2 text-sm">
   132→          <span
   133→            className={`size-2.5 shrink-0 rounded-full ${item.colorClass}`}
   134→            aria-hidden
   135→          />
   136→          <span className="text-muted-foreground">{item.name}</span>
   137→          <span className="font-semibold text-foreground tabular-nums">
   138→            {formatNumber(item.value)}
   139→          </span>
   140→          <span className="text-xs text-muted-foreground tabular-nums">
   141→            ({Math.round((item.value / total) * 100)}%)
   142→          </span>
   143→        </li>
   144→      ))}
   145→    </ul>
   146→  )
   147→}

   698→async function fetchWithdrawals(signal?: AbortSignal): Promise<WithdrawalsResponse> {
   699→  const res = await fetch('/api/withdrawals', { signal })
   700→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)

   704→/** Libellés courts lisibles pour les motifs de retrait (nomenclature brute en majuscules). */

   719→function reasonLabel(reason: string | null | undefined): string {
   720→  if (!reason) return 'Motif non communiqué'
   721→  const mapped = REASON_LABELS[reason]
   722→  if (mapped) return mapped
   723→  return truncate(reason.charAt(0) + reason.slice(1).toLowerCase(), 42)
   724→}
   725→
   726→function reasonBadgeClass(reason: string | null | undefined): string {
   727→  if (!reason) return 'border-border bg-muted/60 text-muted-foreground'
   728→  const r = reason.toUpperCase()
   729→  if (r.includes('MSPRH') || r.includes('PHARMACO')) {
   730→    return 'border-state-danger/30 bg-state-danger/10 text-state-danger'
   731→  }
   732→  if (r.includes('DETENTEUR') || r.includes('LABORATOIRE') || r.includes('COMMERCIAL')) {
   733→    return 'border-state-warning/30 bg-state-warning/10 text-state-warning'
   734→  }
   735→  return 'border-border bg-secondary text-secondary-foreground'
   736→}
