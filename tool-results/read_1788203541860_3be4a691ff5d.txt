    71→async function fetchShortages(signal?: AbortSignal): Promise<ShortagesResponse> {
    72→  const res = await fetch('/api/shortages', { signal })
    73→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)
    74→  return (await res.json()) as ShortagesResponse

    77→async function fetchDrugSuggestions(q: string, signal?: AbortSignal): Promise<DrugSuggestion[]> {
    78→  const res = await fetch(`/api/drugs?q=${encodeURIComponent(q)}&pageSize=8`, { signal })
    79→  if (!res.ok) throw new Error(`Requête échouée (${res.status})`)
    80→  const data = (await res.json()) as { drugs: DrugSuggestion[] }
    81→  return data.drugs
    82→}

    84→async function postShortage(payload: {
    85→  drugId?: number
    86→  brand: string
    87→  dci?: string | null
    88→  wilaya?: string | null
    89→  note?: string | null
    90→}): Promise<void> {
    91→  const res = await fetch('/api/shortages', {
    92→    method: 'POST',
    93→    headers: { 'Content-Type': 'application/json' },
    94→    body: JSON.stringify(payload),
    95→  })
    96→  if (!res.ok) {
    97→    const data = (await res.json().catch(() => null)) as { error?: string } | null
    98→    throw new Error(data?.error ?? 'Échec du signalement')
    99→  }
   100→}

   102→/** Temps relatif en français (« il y a 2 h »). */
   103→function relativeTimeFr(iso: string): string {
   104→  const diff = Date.now() - new Date(iso).getTime()
   105→  const min = Math.floor(diff / 60_000)
   106→  if (min < 1) return "à l'instant"
   107→  if (min < 60) return `il y a ${min} min`
   108→  const h = Math.floor(min / 60)
   109→  if (h < 24) return `il y a ${h} h`
   110→  const j = Math.floor(h / 24)
   111→  if (j < 31) return `il y a ${j} j`
   112→  const mois = Math.floor(j / 30)
   113→  return `il y a ${mois} mois`
   114→}

   116→function useDebounce<T>(value: T, delay: number): T {
   117→  const [debounced, setDebounced] = useState(value)
   118→  useEffect(() => {
   119→    const t = setTimeout(() => setDebounced(value), delay)
   120→    return () => clearTimeout(t)
   121→  }, [value, delay])
   122→  return debounced
   123→}

   135→function DrugAutocomplete({
   136→  selected,
   137→  onSelect,
   138→  onChange,
   139→}: {
   140→  selected: DrugSuggestion | null
   141→  onSelect: (drug: DrugSuggestion | null) => void
   142→  onChange: (value: string) => void
   143→}) {
   144→  const [value, setValue] = useState('')
   145→  const [open, setOpen] = useState(false)
   146→  const wrapperRef = useRef<HTMLDivElement>(null)
   147→  const debounced = useDebounce(value, 300)
   148→  const trimmed = debounced.trim()
   149→
   150→  const { data: results, isFetching } = useQuery({
   151→    queryKey: ['shortage-drug-search', trimmed],
   152→    queryFn: ({ signal }) => fetchDrugSuggestions(trimmed, signal),
   153→    enabled: trimmed.length >= 2,
   154→    placeholderData: keepPreviousData,
   155→  })
   156→
   157→  useEffect(() => {
   158→    function onDocMouseDown(e: MouseEvent) {
   159→      if (wrapperRef.current && !wrapperRef.current.contains(e.target as Node)) {
   160→        setOpen(false)
   161→      }
   162→    }
   163→    document.addEventListener('mousedown', onDocMouseDown)
   164→    return () => document.removeEventListener('mousedown', onDocMouseDown)
   165→  }, [])
   166→
   167→  const isOpen = open && trimmed.length >= 2 && !selected
   168→
   169→  function handleSelect(drug: DrugSuggestion) {
   170→    onSelect(drug)
   171→    setValue('')
   172→    setOpen(false)
   173→    onChange(drug.brand)
   174→  }
   175→
   176→  return (
   177→    <div ref={wrapperRef} className="relative">
   178→      <div
   179→        className={cn(
   180→          'flex items-center gap-2 rounded-xl border bg-card transition-colors',
   181→          'border-border focus-within:border-primary/60 focus-within:ring-4 focus-within:ring-primary/20'
   182→        )}
   183→      >
   184→        {selected ? (
   185→          <div className="flex h-11 w-full items-center justify-between gap-2 rounded-xl bg-primary/10 px-3">
   186→            <span className="min-w-0">
   187→              <span className="block truncate text-sm font-semibold text-foreground">
   188→                {selected.brand}
   189→              </span>
   190→              <span className="block truncate text-xs text-muted-foreground">
   191→                {selected.dci}
   192→                {selected.dosage ? ` · ${selected.dosage}` : ''}
   193→              </span>
   194→            </span>
   195→            <Button
   196→              type="button"
   197→              variant="ghost"
   198→              size="icon"
   199→              className="size-8 shrink-0 rounded-lg text-muted-foreground hover:text-foreground"
   200→              onClick={() => {
   201→                onSelect(null)
   202→                onChange('')
   203→              }}
   204→              aria-label="Effacer le médicament sélectionné"
   205→            >
   206→              <X className="size-4" aria-hidden />
   207→            </Button>
   208→          </div>
   209→        ) : (
   210→          <Input
   211→            type="text"
   212→            role="combobox"
   213→            aria-expanded={isOpen}
   214→            aria-autocomplete="list"
   215→            aria-label="Nom du médicament en pénurie"
   216→            placeholder="Ex : DOLIPRANE, AUGMENTIN…"
   217→            value={value}
   218→            onChange={(e) => {
   219→              setValue(e.target.value)
   220→              onChange(e.target.value)
   221→              setOpen(true)
   222→            }}
   223→            onFocus={() => setOpen(true)}
   224→            autoComplete="off"
   225→            spellCheck={false}
   226→            className="h-11 border-0 shadow-none focus-visible:ring-0 focus-visible:ring-offset-0"
   227→          />
   228→        )}
   229→        {isFetching && isOpen ? (
   230→          <Loader2 className="mr-3 size-4 shrink-0 animate-spin text-muted-foreground" aria-hidden />
   231→        ) : null}
   232→      </div>
   233→
   234→      {isOpen ? (
   235→        <div
   236→          role="listbox"
   237→          aria-label="Suggestions de médicaments"
   238→          className="scroll-thin absolute inset-x-0 top-full z-50 mt-2 max-h-64 overflow-y-auto rounded-xl border border-border bg-popover p-1.5 shadow-xl shadow-black/10"
   239→        >
   240→          {!results || results.length === 0 ? (
   241→            <p className="px-3 py-5 text-center text-sm text-muted-foreground">
   242→              {isFetching ? 'Recherche…' : `Aucun médicament trouvé pour « ${trimmed} »`}
   243→            </p>
   244→          ) : (
   245→            results.map((drug) => (
   246→              <button
   247→                key={drug.id}
   248→                type="button"
   249→                role="option"
   250→                aria-selected={false}
   251→                onClick={() => handleSelect(drug)}
   252→                className="flex w-full items-center justify-between gap-3 rounded-lg px-3 py-2.5 text-left transition-colors hover:bg-accent"
   253→              >
   254→                <span className="min-w-0">
   255→                  <span className="block truncate text-sm font-semibold text-foreground">
   256→                    {drug.brand}
   257→                  </span>
   258→                  <span className="block truncate text-xs text-muted-foreground">
   259→                    {drug.dci}
   260→                    {drug.dosage ? ` · ${drug.dosage}` : ''}
   261→                  </span>
   262→                </span>
   263→                <span
   264→                  className={cn(
   265→                    'flex shrink-0 items-center gap-1.5 rounded-md border border-border bg-muted/60 px-2 py-0.5 text-[10px] font-medium text-muted-foreground',
   266→                  )}
   267→                >
   268→                  <span
   269→                    className={cn('size-1.5 rounded-full', STATUS_DOT[drug.status] ?? 'bg-muted-foreground')}
   270→                    aria-hidden
   271→                  />
   272→                  {drug.status === 'ACTIF'
   273→                    ? 'Actif'
   274→                    : drug.status === 'RETRIE'
   275→                      ? 'Retiré'
   276→                      : drug.status === 'NON_RENOUVELE'
   277→                        ? 'Non renouvelé'
   278→                        : drug.status}
   279→                </span>
   280→              </button>
   281→            ))
   282→          )}
   283→        </div>
   284→      ) : null}
   285→    </div>
   286→  )
   287→}
