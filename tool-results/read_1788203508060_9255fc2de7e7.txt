   192→function useCountUp(target: number, duration = 900): number {
   193→  const [value, setValue] = useState(0)
   194→  useEffect(() => {
   195→    if (target <= 0) return
   196→    let raf = 0
   197→    const start = performance.now()
   198→    const tick = (now: number) => {
   199→      const p = Math.min(1, (now - start) / duration)
   200→      const eased = 1 - Math.pow(1 - p, 3)
   201→      setValue(Math.round(target * eased))
   202→      if (p < 1) raf = requestAnimationFrame(tick)
   203→    }
   204→    raf = requestAnimationFrame(tick)
   205→    return () => cancelAnimationFrame(raf)
   206→  }, [target, duration])
   207→  return value
   208→}
