   141→function VerifierPanel() {
   142→  const basket = useDzPharm((s) => s.basket)
   143→  const addToBasket = useDzPharm((s) => s.addToBasket)
   144→  const removeFromBasket = useDzPharm((s) => s.removeFromBasket)
   145→  const clearBasket = useDzPharm((s) => s.clearBasket)
   146→  const openDrug = useDzPharm((s) => s.openDrug)
   147→  const { toast } = useToast()
   148→
   149→  const [patientContext, setPatientContext] = useState('')
   150→  const [result, setResult] = useState<InteractionsResponse | null>(null)
   151→  const [analyzedIds, setAnalyzedIds] = useState<string>('')
   152→  const [localPending, setLocalPending] = useState(false)
   153→  const [aiPending, setAiPending] = useState(false)
   154→  const [aiFailed, setAiFailed] = useState(false)
   155→
   156→  const names = useMemo(() => basket.map((b) => b.brand), [basket])
   157→
   158→  async function analyze() {
   159→    if (names.length < 2) return
   160→    setAiFailed(false)
   161→
   162→    // 1. Moteur local — instantané, toujours disponible
   163→    setLocalPending(true)
   164→    try {
   165→      const local = await postLocalInteractions(names)
   166→      setResult(local)
   167→      setAnalyzedIds(basket.map((b) => b.id).join(','))
   168→    } catch {
   169→      // le moteur local échoue rarement ; on tente quand même l'IA
   170→    } finally {
   171→      setLocalPending(false)
   172→    }
   173→
   174→    // 2. Analyse IA approfondie — remplace le résultat local si disponible
   175→    setAiPending(true)
   176→    try {
   177→      const ai = await postInteractions(names, patientContext)
   178→      setResult({ ...ai, source: 'ai' })
   179→      setAnalyzedIds(basket.map((b) => b.id).join(','))
   180→    } catch {
   181→      setAiFailed(true)
   182→    } finally {
   183→      setAiPending(false)
   184→    }
   185→  }
   186→
   187→  const pending = localPending || aiPending
   188→  const stale =
   189→    result !== null && analyzedIds !== basket.map((b) => b.id).join(',')
   190→
   191→  const severityCounts = useMemo(() => {
   192→    const counts = new Map<InteractionSeverity, number>()
   193→    for (const p of result?.pairs ?? []) {
   194→      counts.set(p.severity, (counts.get(p.severity) ?? 0) + 1)
   195→    }
   196→    return counts
   197→  }, [result])
   198→
   199→  function handleSelect(name: Parameters<typeof addToBasket>[0]) {
   200→    const res = addToBasket(name)
   201→    if (res === 'added') {
   202→      toast({
   203→        title: 'Médicament ajouté',
   204→        description: `${name.brand} (${name.dci}) — ${basket.length + 1} produit(s) dans le panier.`,
   205→      })
   206→    } else if (res === 'duplicate') {
   207→      toast({ title: 'Déjà présent', description: `${name.brand} est déjà dans le panier.` })
   208→    } else {
   209→      toast({
   210→        title: 'Panier complet',
   211→        description: `Le contrôle est limité à ${MAX_BASKET} médicaments.`,
   212→        variant: 'destructive',
   213→      })
   214→    }
   215→  }
   216→
   217→  const risk = result ? RISK_META[result.globalRisk] : null
   218→
   219→  return (
   220→    <div className="grid grid-cols-1 gap-6 lg:grid-cols-[380px_1fr] xl:grid-cols-[420px_1fr]">
   221→      {/* ------------------------- Panier ------------------------- */}
   222→      <div className="space-y-4">
   223→        <Card>
   224→          <CardHeader className="pb-3">
   225→            <CardTitle className="flex items-center justify-between text-base">
   226→              <span className="flex items-center gap-2">
   227→                <Stethoscope className="size-4 text-primary" aria-hidden />
   228→                Panier d&apos;analyse
   229→              </span>
   230→              <span className="rounded-full bg-primary/10 px-2 py-0.5 text-xs font-semibold text-primary tabular-nums">
   231→                {basket.length}/{MAX_BASKET}
   232→              </span>
   233→            </CardTitle>
   234→          </CardHeader>
   235→          <CardContent className="space-y-4">
   236→            <SearchAutocomplete
   237→              placeholder="Ajouter un médicament…"
   238→              ariaLabel="Ajouter un médicament au contrôle d'interactions"
   239→              onSelect={handleSelect}
   240→              rightHint={
   241→                <Plus className="size-4 shrink-0 text-muted-foreground/50" aria-hidden />
   242→              }
   243→            />
   244→
   245→            {basket.length === 0 ? (
   246→              <p className="rounded-lg border border-dashed border-border p-4 text-center text-sm text-muted-foreground">
   247→                Le panier est vide. Recherchez un médicament pour l&apos;ajouter —
   248→                depuis le répertoire, utilisez le bouton «&nbsp;Ajouter au contrôle
   249→                d&apos;interactions&nbsp;».
   250→              </p>
   251→            ) : (
   252→              <ul className="flex flex-wrap gap-2" aria-label="Médicaments sélectionnés">
   253→                {basket.map((item) => (
   254→                  <li key={item.id}>
   255→                    <span className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-secondary/60 py-1.5 pr-1.5 pl-3 text-sm">
   256→                      <button
   257→                        type="button"
   258→                        onClick={() => openDrug(item.id)}
   259→                        className="min-w-0 text-left focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   260→                        title="Ouvrir la fiche du médicament"
   261→                      >
   262→                        <span className="block truncate font-medium text-foreground">
   263→                          {item.brand}
   264→                        </span>
   265→                        <span className="block truncate text-[11px] text-muted-foreground">
   266→                          {item.dci}
   267→                        </span>
   268→                      </button>
   269→                      <button
   270→                        type="button"
   271→                        onClick={() => removeFromBasket(item.id)}
   272→                        aria-label={`Retirer ${item.brand} du panier`}
   273→                        className="flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-state-danger/10 hover:text-state-danger focus-visible:ring-2 focus-visible:ring-ring focus-visible:outline-none"
   274→                      >
   275→                        <X className="size-3.5" />
   276→                      </button>
   277→                    </span>
   278→                  </li>
   279→                ))}
   280→              </ul>
   281→            )}
   282→
   283→            <div>
   284→              <label
   285→                htmlFor="patient-context"
   286→                className="mb-1.5 block text-xs font-medium text-muted-foreground"
   287→              >
   288→                Contexte patient — âge, pathologies, grossesse…
   289→              </label>
   290→              <Textarea
   291→                id="patient-context"
   292→                value={patientContext}
   293→                onChange={(e) => setPatientContext(e.target.value)}
   294→                placeholder="Ex : femme enceinte 2e trimestre, insuffisance rénale chronique, 72 ans…"
   295→                rows={3}
   296→                className="resize-none text-sm"
   297→              />
   298→            </div>
   299→
   300→            <div className="flex gap-2">
   301→              <Button
   302→                onClick={analyze}
   303→                disabled={basket.length < 2 || pending}
   304→                className="h-11 flex-1 text-sm font-semibold"
   305→              >
   306→                {pending ? (
   307→                  <>
   308→                    <Loader2 className="size-4 animate-spin" aria-hidden />
   309→                    {localPending ? 'Règles locales…' : 'Analyse IA…'}
   310→                  </>
   311→                ) : (
   312→                  <>
   313→                    <ShieldAlert className="size-4" aria-hidden />
   314→                    Analyser les interactions
   315→                  </>
   316→                )}
   317→              </Button>
   318→              {basket.length > 0 ? (
   319→                <Button
   320→                  variant="ghost"
   321→                  size="icon"
   322→                  className="size-11 text-muted-foreground hover:text-state-danger"
   323→                  onClick={() => {
   324→                    clearBasket()
   325→                    setResult(null)
   326→                  }}
   327→                  aria-label="Vider le panier"
   328→                >
   329→                  <Trash2 className="size-4" />
   330→                </Button>
   331→              ) : null}
   332→            </div>
   333→            {pending ? (
   334→              <p className="flex items-center justify-center gap-2 text-center text-xs text-muted-foreground">
   335→                <Loader2 className="size-3 animate-spin" aria-hidden />
   336→                {localPending
   337→                  ? 'Vérification instantanée par le moteur local de règles…'
   338→                  : 'L\u2019analyse IA approfondie peut prendre jusqu\u2019à une minute…'}
   339→              </p>
   340→            ) : null}
   341→          </CardContent>
   342→        </Card>
   343→      </div>
   344→
   345→      {/* ------------------------ Résultats ------------------------ */}
   346→      <div className="min-w-0">
   347→        {!result && !pending ? (
   348→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-dashed border-border p-10 text-center">
   349→            <ShieldCheck className="size-12 text-muted-foreground/40" aria-hidden />
   350→            <p className="mt-4 text-base font-semibold text-foreground">
   351→              Aucune analyse pour le moment
   352→            </p>
   353→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   354→              Ajoutez au moins 2 médicaments au panier puis lancez l&apos;analyse. Le
   355→              rapport présente le niveau de risque global, les associations à
   356→              surveiller et les conseils de dispensation.
   357→            </p>
   358→          </div>
   359→        ) : pending && !result ? (
   360→          <div className="flex h-full min-h-[320px] flex-col items-center justify-center rounded-xl border border-border bg-card p-10 text-center">
   361→            <Loader2 className="size-10 animate-spin text-primary" aria-hidden />
   362→            <p className="mt-4 text-base font-semibold text-foreground">
   363→              Analyse en cours…
   364→            </p>
   365→            <p className="mt-1 max-w-md text-sm text-muted-foreground">
   366→              Le moteur local de règles vérifie instantanément les {basket.length}{' '}
   367→              médicaments de votre panier, puis l&apos;IA approfondit l&apos;analyse.
   368→            </p>
   369→          </div>
   370→        ) : result ? (
   371→          <div className="space-y-4">
   372→            {/* Indicateur de source + état IA */}
   373→            <div
   374→              className={cn(
   375→                'flex flex-wrap items-center gap-2 rounded-lg border px-4 py-2.5 text-sm',
   376→                aiFailed
   377→                  ? 'border-state-warning/40 bg-state-warning/10 text-state-warning'
   378→                  : 'border-border bg-card text-muted-foreground'
   379→              )}
   380→              role="status"
   381→            >
   382→              {result.source === 'ai' ? (
   383→                <>
   384→                  <Sparkles className="size-4 shrink-0 text-primary" aria-hidden />
   385→                  <span>
   386→                    Analyse IA approfondie — croisée avec la base de règles locale.
   387→                  </span>
   388→                </>
   389→              ) : aiFailed ? (
   390→                <>
   391→                  <AlertTriangle className="size-4 shrink-0" aria-hidden />
   392→                  <span>
   393→                    Analyse IA approfondie indisponible — résultats du{' '}
   394→                    <strong className="font-semibold">moteur local de règles</strong>{' '}
   395→                    ({result.pairs.length} association(s) vérifiée(s)).
   396→                  </span>
   397→                </>
   398→              ) : aiPending ? (
   399→                <>
   400→                  <Loader2 className="size-4 shrink-0 animate-spin" aria-hidden />
   401→                  <span>Règles locales appliquées — analyse IA en cours…</span>
   402→                </>
   403→              ) : (
   404→                <>
   405→                  <Zap className="size-4 shrink-0 text-primary" aria-hidden />
   406→                  <span>
   407→                    Moteur local de règles — réponse instantanée hors ligne.
   408→                  </span>
   409→                </>
   410→              )}
   411→            </div>
   412→
   413→            {stale ? (
   414→              <div
   415→                className="flex items-center gap-2 rounded-lg border border-state-warning/40 bg-state-warning/10 px-4 py-2.5 text-sm text-state-warning"
   416→                role="status"
   417→              >
   418→                <AlertTriangle className="size-4 shrink-0" aria-hidden />
   419→                Le panier a été modifié depuis cette analyse — relancez l&apos;analyse
   420→                pour actualiser les résultats.
   421→              </div>
   422→            ) : null}
   423→
   424→            {/* Bandeau risque global */}
   425→            <div
   426→              className={cn(
   427→                'flex items-start gap-4 rounded-xl border p-5',
   428→                risk?.border,
   429→                risk?.className
   430→              )}
   431→              role="status"
   432→              aria-live="polite"
   433→            >
   434→              {result.globalRisk === 'FAIBLE' ? (
   435→                <ShieldCheck className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   436→              ) : (
   437→                <ShieldAlert className={cn('size-9 shrink-0', risk?.icon)} aria-hidden />
   438→              )}
   439→              <div className="min-w-0">
   440→                <p className="text-lg font-bold tracking-tight">
   441→                  {risk?.label}
   442→                  {result.pairs.length > 0 ? (
   443→                    <span className="ml-2 align-middle text-sm font-medium opacity-80">
   444→                      · {result.pairs.length} association{result.pairs.length > 1 ? 's' : ''} analysée
   445→                      {result.pairs.length > 1 ? 's' : ''}
   446→                    </span>
   447→                  ) : null}
   448→                </p>
   449→                <p className="mt-1 text-sm leading-relaxed text-foreground/90">{result.summary}</p>
   450→              </div>
   451→            </div>
   452→
   453→            {/* Répartition des gravités */}
   454→            {result.pairs.length > 0 ? (
   455→              <div className="flex flex-wrap items-center gap-x-5 gap-y-2 rounded-lg border border-border bg-card px-4 py-2.5">
   456→                {SEVERITY_ORDER.filter((s) => severityCounts.has(s)).map((s) => (
   457→                  <span
   458→                    key={s}
   459→                    className="flex items-center gap-1.5 text-xs font-medium text-muted-foreground"
   460→                  >
   461→                    <span className={cn('size-2 rounded-full', SEVERITY_LEGEND[s])} aria-hidden />
   462→                    {SEVERITY_LABELS[s]}
   463→                    <span className="font-semibold text-foreground tabular-nums">
   464→                      {severityCounts.get(s)}
   465→                    </span>
   466→                  </span>
   467→                ))}
   468→              </div>
   469→            ) : null}
   470→
   471→            {/* Médicaments enrichis */}
   472→            {result.enriched.length > 0 ? (
   473→              <Card>
   474→                <CardHeader className="pb-2">
   475→                  <CardTitle className="text-sm">Produits analysés</CardTitle>
   476→                </CardHeader>
   477→                <CardContent className="flex flex-wrap gap-2">
   478→                  {result.enriched.map((e) => (
   479→                    <span
   480→                      key={e.input}
   481→                      className="flex max-w-full items-center gap-2 rounded-lg border border-border bg-muted/40 px-3 py-1.5 text-xs"
   482→                    >
   483→                      <span className="min-w-0">
   484→                        <span className="block truncate font-semibold text-foreground">
   485→                          {e.input}
   486→                        </span>
   487→                        {e.dci ? (
   488→                          <span className="block truncate text-muted-foreground">{e.dci}</span>
   489→                        ) : null}
   490→                      </span>
   491→                      {e.status ? (
   492→                        e.status === 'RETRIE' ? (
   493→                          <span className="shrink-0 rounded bg-state-danger/15 px-1.5 py-0.5 text-[10px] font-semibold text-state-danger">
   494→                            Retiré
   495→                          </span>
   496→                        ) : (
   497→                          <StatusBadge status={e.status} className="shrink-0" />
   498→                        )
   499→                      ) : null}
   500→                    </span>
   501→                  ))}
   502→                </CardContent>
   503→              </Card>
   504→            ) : null}
   505→
   506→            {/* Paires */}
   507→            {result.pairs.length > 0 ? (
   508→              <section aria-label="Interactions détectées" className="space-y-3">
   509→                <h2 className="text-sm font-semibold tracking-wider text-muted-foreground uppercase">
   510→                  Interactions détectées
   511→                </h2>
   512→                {result.pairs.map((pair, i) => (
   513→                  <PairCard key={`${pair.drugs.join('-')}-${i}`} pair={pair} />
   514→                ))}
   515→              </section>
   516→            ) : (
   517→              <Card className="border-state-safe/40 bg-state-safe/5">
   518→                <CardContent className="flex items-center gap-3 p-4 text-sm text-foreground/90">
   519→                  <ShieldCheck className="size-5 shrink-0 text-state-safe" aria-hidden />
   520→                  Aucune interaction significative identifiée entre les produits analysés.
   521→                </CardContent>
   522→              </Card>
   523→            )}
