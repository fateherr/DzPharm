   465→export function normalizeKey(s: string): string {
   466→  return s
   467→    .normalize('NFD')
   468→    .replace(/[\u0300-\u036f]/g, '')
   469→    .toUpperCase()
   470→    .replace(/[^A-Z0-9 ]/g, ' ')
   471→    .replace(/\s+/g, ' ')
   472→    .trim()
   473→}

   475→/**
   476→ * Détecte les règles applicables sur une liste de médicaments.
   477→ * Chaque médicament est identifié par sa marque (`name` ou `input`) et sa DCI
   478→ * (l'une des deux peut manquer) ; la correspondance se fait sur les jetons
   479→ * DCI/marque normalisés.
   480→ */
   481→export function matchInteractions(
   482→  drugs: { name?: string; input?: string; dci?: string | null }[]
   483→): LocalRule[] {
   484→  const keysPerDrug = drugs.map((d) => {
   485→    const label = d.name ?? d.input ?? ''
   486→    const keys = new Set<string>()
   487→    if (d.dci) {
   488→      const k = normalizeKey(d.dci)
   489→      if (k) keys.add(k)
   490→      // les DCI composées (« X + Y ») sont éclatées
   491→      for (const part of k.split(/(?: \+ | ET )/)) {
   492→        const p = part.trim()
   493→        if (p.length >= 3) keys.add(p)
   494→      }
   495→    }
   496→    const n = normalizeKey(label)
   497→    if (n) keys.add(n)
   498→    return Array.from(keys)
   499→  })
   500→
   501→  const allKeys = new Set(keysPerDrug.flat())
   502→
   503→  return LOCAL_RULES.filter((rule) => {
   504→    const leftHit = rule.left.some((t) => tokenMatch(allKeys, t))
   505→    const rightHit = rule.right.some((t) => tokenMatch(allKeys, t))
   506→    // les deux membres doivent être touchés par des médicaments distincts
   507→    if (!leftHit || !rightHit) return false
   508→    for (let i = 0; i < drugs.length; i++) {
   509→      const di = keysPerDrug[i]
   510→      const leftSelf = rule.left.some((t) => tokenMatch(new Set(di), t))
   511→      const rightSelf = rule.right.some((t) => tokenMatch(new Set(di), t))
   512→      if (leftSelf && rightSelf) {
   513→        // un seul produit couvre les deux membres (ex : association fixe) → non pertinent
   514→        continue
   515→      }
   516→      for (let j = i + 1; j < drugs.length; j++) {
   517→        const dj = keysPerDrug[j]
   518→        const pairLeft = rule.left.some((t) => tokenMatch(new Set(di), t))
   519→        const pairRight = rule.right.some((t) => tokenMatch(new Set(dj), t))
   520→        const pairLeftR = rule.right.some((t) => tokenMatch(new Set(di), t))
   521→        const pairRightL = rule.left.some((t) => tokenMatch(new Set(dj), t))
   522→        if ((pairLeft && pairRight) || (pairLeftR && pairRightL)) return true
   523→      }
   524→    }
   525→    return false
   526→  })
   527→}
   528→
   529→function tokenMatch(keys: Set<string>, token: string): boolean {
   530→  if (keys.has(token)) return true
   531→  const t = ` ${token} `
   532→  for (const k of keys) {
   533→    if (k === token) return true
   534→    if ((` ${k} `).includes(t) && token.length >= 4) return true
   535→    if ((` ${token} `).includes(` ${k} `) && k.length >= 4) return true
   536→  }
   537→  return false
   538→}

   540→export const SEVERITY_WEIGHT: Record<LocalSeverity, number> = {
   541→  'CONTRE-INDIQUE': 4,
   542→  MAJEURE: 3,
   543→  MODEREE: 2,
   544→  MINEURE: 1,
   545→}
   546→
   547→export function globalRiskFromPairs(
   548→  pairs: { severity: LocalSeverity }[]
   549→): 'FAIBLE' | 'MODERE' | 'ELEVE' | 'CRITIQUE' {
   550→  if (pairs.length === 0) return 'FAIBLE'
   551→  const max = Math.max(...pairs.map((p) => SEVERITY_WEIGHT[p.severity]))
   552→  if (max >= 4) return 'CRITIQUE'
   553→  if (max === 3) return 'ELEVE'
   554→  if (max === 2) return 'MODERE'
   555→  return 'FAIBLE'
   556→}

   558→/** Associe une règle aux paires de produits concernés (marques lisibles). */
   559→export function pairsFromRule(
   560→  rule: LocalRule,
   561→  drugs: { name?: string; input?: string; dci?: string | null }[]
   562→): [string, string][] {
   563→  const result: [string, string][] = []
   564→  for (let i = 0; i < drugs.length; i++) {
   565→    for (let j = i + 1; j < drugs.length; j++) {
   566→      const ki = new Set(
   567→        [drugs[i].dci, drugs[i].name ?? drugs[i].input]
   568→          .filter(Boolean)
   569→          .map((s) => normalizeKey(s as string))
   570→      )
   571→      const kj = new Set(
   572→        [drugs[j].dci, drugs[j].name ?? drugs[j].input]
   573→          .filter(Boolean)
   574→          .map((s) => normalizeKey(s as string))
   575→      )
   576→      const iL = rule.left.some((t) => tokenMatch(ki, t))
   577→      const iR = rule.right.some((t) => tokenMatch(ki, t))
   578→      const jL = rule.left.some((t) => tokenMatch(kj, t))
   579→      const jR = rule.right.some((t) => tokenMatch(kj, t))
   580→      if ((iL && jR) || (iR && jL)) {
   581→        const label = (d: { name?: string; input?: string; dci?: string | null }) =>
   582→          d.name ?? d.input ?? d.dci ?? '—'
   583→        result.push([label(drugs[i]), label(drugs[j])])
   584→      }
   585→    }
   586→  }
   587→  return result
   588→}
